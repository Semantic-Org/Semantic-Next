### Component Communication

Connect components without complex state management:

```javascript
// Child finds parent automatically
const parent = findParent('shopping-app');
parent.updateCartCount(items.length);

// Parent manages children
const cartComponent = findChild('shopping-cart');
cartComponent.addItem(product);

// Event communication
dispatchEvent('productAdded', { product, quantity });
```

### Real-World Example: E-commerce App

```javascript
defineComponent({
  tagName: 'product-card',
  template: `
    <img src="{image}" alt="{name}">
    <h3>{name}</h3>
    <div class="price">${{price}}</div>
    <button onclick="{{() => self.addToCart()}}" 
            class="add-btn {inCart ? 'added' : ''}">
      {inCart ? 'Added ✓' : 'Add to Cart'}
    </button>
  `,
  defaultSettings: {
    productId: '',
    name: '',
    price: 0,
    image: ''
  },
  defaultState: {
    inCart: false
  },
  createComponent: ({ state, settings, self }) => ({
    addToCart() {
      // Find parent cart
      const cart = findParent('shopping-cart');
      
      // Add product
      cart.addProduct({
        id: settings.productId,
        name: settings.name,
        price: settings.price
      });
      
      // Update local state
      state.inCart.set(true);
      
      // Notify other components
      dispatchEvent('productAdded', {
        productId: settings.productId
      });
    }
  })
});
```
